  // Reverse-Joint (Digitigrade) 2-Bone IK Solver for Mech Legs
  function solveReverseLegIK(hip, foot, l1, l2) {
    const dx = foot.x - hip.x;
    const dy = foot.y - hip.y;
    let dist = Math.hypot(dx, dy);
    dist = Math.min(Math.max(dist, 15), l1 + l2 - 0.5);

    const cosAngle2 = (dist * dist - l1 * l1 - l2 * l2) / (2 * l1 * l2);
    const angle2 = -Math.acos(Math.min(Math.max(cosAngle2, -1), 1)); // Reverse knee bend

    const baseAngle = Math.atan2(dy, dx);
    const cosAngle1 = (dist * dist + l1 * l1 - l2 * l2) / (2 * dist * l1);
    const angle1 = baseAngle + Math.acos(Math.min(Math.max(cosAngle1, -1), 1));

    return {
      thighAngleDeg: toDeg(angle1),
      shinAngleDeg: toDeg(angle1 + angle2),
      kneePos: {
        x: hip.x + Math.cos(angle1) * l1,
        y: hip.y + Math.sin(angle1) * l1
      }
    };
  }

  // Exact center-rotation translation compensation
  function getRotatedBoneBounds(jointX, jointY, length, height, angleDeg) {
    const rad = toRad(angleDeg);
    const halfL = length / 2;
    const halfH = height / 2;
    const centerX = jointX + halfL * Math.cos(rad);
    const centerY = jointY + halfL * Math.sin(rad);

    return {
      left: round2(centerX - halfL),
      top: round2(centerY - halfH),
      rotation: round2(angleDeg)
    };
  }

  // Hermite Spline Evaluator for Aerial Drone Flight
  function evaluateDronePath(t01) {
    const pts = CONFIG.drone.path;
    for (let i = 0; i < pts.length - 1; i++) {
      const p1 = pts[i];
      const p2 = pts[i + 1];
      if (t01 >= p1.t && t01 <= p2.t) {
        const u = (t01 - p1.t) / (p2.t - p1.t);
        const easeU = u * u * (3 - 2 * u);
        return {
          x: p1.x + (p2.x - p1.x) * easeU,
          y: p1.y + (p2.y - p1.y) * easeU,
          bank: p1.bank + (p2.bank - p1.bank) * easeU
        };
      }
    }
    return pts[pts.length - 1];
  }

  // ==========================================================================
  // 3. COMBAT ENCOUNTER EVALUATOR (CHARACTERS & KINEMATICS)
  // ==========================================================================
  function generateCombatTimeline(ids) {
    const layers = [];
    const makeTrack = (propertyKind, keyframes) => ({ propertyKind, durationSeconds: CONFIG.duration, keyframes });

    // Mech Buffers
    const pelvisX = [], pelvisY = [], pelvisRot = [];
    const hullX = [], hullY = [], hullRot = [];
    const eyeX = [], eyeY = [];
    const podX = [], podY = [], podRot = [];
    const cannonX = [], cannonY = [], cannonRot = [];
    const cannonHubX = [], cannonHubY = [];
    const flashX = [], flashY = [], flashScale = [], flashOpacity = [];
    const shieldX = [], shieldY = [], shieldScale = [], shieldOpacity = [];
    const sparksX = [], sparksY = [], sparksScale = [], sparksOpacity = [];
    const missileX = [], missileY = [], missileRot = [], missileOpacity = [];

    // Mech Legs
    const legL = { thX: [], thY: [], thRot: [], kneeX: [], kneeY: [], shX: [], shY: [], shRot: [], ftX: [], ftY: [], ftRot: [] };
    const legR = { thX: [], thY: [], thRot: [], kneeX: [], kneeY: [], shX: [], shY: [], shRot: [], ftX: [], ftY: [], ftRot: [] };
    const shadowX = [], shadowY = [], shadowScaleX = [];

    // Drone Buffers
    const droneFusX = [], droneFusY = [], droneFusRot = [];
    const droneVisX = [], droneVisY = [];
    const thrustLX = [], thrustLY = [], thrustRX = [], thrustRY = [];
    const laserX = [], laserY = [], laserRot = [], laserScaleX = [], laserOpacity = [];
    const flareX = [], flareY = [], flareScale = [], flareOpacity = [];

    // Mech Foot Stance Solver
    function getMechFootTrajectory(phaseAngle, rootHipX) {
      const pNorm = (phaseAngle % (Math.PI * 2) + Math.PI * 2) % (Math.PI * 2);
      const u = pNorm / (Math.PI * 2);
      let fx, fy, fRot;

      if (u <= 0.5) {
        const stanceT = u / 0.5;
        fx = rootHipX + 42 * (0.5 - stanceT);
        fy = CONFIG.mech.groundY;
        fRot = stanceT < 0.15 ? -14 * (1 - stanceT / 0.15) : (stanceT > 0.70 ? 20 * ((stanceT - 0.70) / 0.30) : 0);
      } else {
        const swingT = (u - 0.5) / 0.5;
        fx = rootHipX + (-0.5 + Math.sin(swingT * Math.PI * 0.5)) * 42;
        fy = CONFIG.mech.groundY - Math.sin(swingT * Math.PI) * 22;
        fRot = swingT < 0.4 ? 18 * (1 - swingT / 0.4) : -4;
      }
      return { x: fx, y: fy, rot: fRot };
    }

    for (let f = 0; f <= totalFrames; f++) {
      const t = f / totalFrames;
      const walkPhase = t * 2 * Math.PI * 2; // 2 heavy forward strides

      // ----------------------------------------------------------------------
      // STEP 1: EVALUATE DRONE FLIGHT TRAJECTORY (BEHAVIOR STATE MACHINE)
      // ----------------------------------------------------------------------
      const dronePose = evaluateDronePath(t);
      const droneRad = toRad(dronePose.bank);

      droneFusX.push({ normalizedTime: t, value: { number: round2(dronePose.x - CONFIG.drone.width / 2) } });
      droneFusY.push({ normalizedTime: t, value: { number: round2(dronePose.y - CONFIG.drone.height / 2) } });
      droneFusRot.push({ normalizedTime: t, value: { number: round2(dronePose.bank) } });

      // Drone Optic Sensor
      const visOffset = -CONFIG.drone.width / 2 + 6;
      droneVisX.push({ normalizedTime: t, value: { number: round2(dronePose.x + Math.cos(droneRad) * visOffset - 5) } });
      droneVisY.push({ normalizedTime: t, value: { number: round2(dronePose.y + Math.sin(droneRad) * visOffset - 5) } });

      // Thrusters
      thrustLX.push({ normalizedTime: t, value: { number: round2(dronePose.x + 14) } });
      thrustLY.push({ normalizedTime: t, value: { number: round2(dronePose.y - 8) } });
      thrustRX.push({ normalizedTime: t, value: { number: round2(dronePose.x + 14) } });
      thrustRY.push({ normalizedTime: t, value: { number: round2(dronePose.y + 4) } });

      // ----------------------------------------------------------------------
      // STEP 2: MECH KINEMATICS & DYNAMIC LOOK-AT AUTOCANNON TRACKING
      // ----------------------------------------------------------------------
      const currentX = (t < 0.6)
        ? CONFIG.mech.startX + (t / 0.6) * (CONFIG.mech.endX - CONFIG.mech.startX)
        : CONFIG.mech.endX; // Holds combat stance

      const bounceY = (t < 0.6) ? -Math.sin(walkPhase * 2) * 5.0 : 0;
      const currentHipY = CONFIG.mech.pelvisBaseY + bounceY;

      // Pelvis & Hull
      pelvisX.push({ normalizedTime: t, value: { number: round2(currentX - 17) } });
      pelvisY.push({ normalizedTime: t, value: { number: round2(currentHipY - 10) } });
      pelvisRot.push({ normalizedTime: t, value: { number: round2((t < 0.6) ? Math.sin(walkPhase) * 3 : 0) } });

      const hullCenterY = currentHipY - 40;
      hullX.push({ normalizedTime: t, value: { number: round2(currentX - 42) } });
      hullY.push({ normalizedTime: t, value: { number: round2(hullCenterY - 23) } });
      hullRot.push({ normalizedTime: t, value: { number: 0 } });

      eyeX.push({ normalizedTime: t, value: { number: round2(currentX + 14) } });
      eyeY.push({ normalizedTime: t, value: { number: round2(hullCenterY - 6) } });

      // Missile Pod
      podX.push({ normalizedTime: t, value: { number: round2(currentX - 36) } });
      podY.push({ normalizedTime: t, value: { number: round2(hullCenterY - 26) } });
      podRot.push({ normalizedTime: t, value: { number: -18 } }); // Tilted upward

      // Autocannon solves continuous LookAt(dronePose)
      const cannonPivotX = currentX + 16;
      const cannonPivotY = hullCenterY + 4;
      const aimDx = dronePose.x - cannonPivotX;
      const aimDy = dronePose.y - cannonPivotY;
      const aimAngleDeg = toDeg(Math.atan2(aimDy, aimDx));

      cannonHubX.push({ normalizedTime: t, value: { number: round2(cannonPivotX - 10) } });
      cannonHubY.push({ normalizedTime: t, value: { number: round2(cannonPivotY - 10) } });

      // Mech fires flak burst at t = [0.55, 0.70]
      let isMechFiring = (t >= 0.55 && t <= 0.70);
      let cannonRecoil = isMechFiring ? -Math.abs(Math.sin((t - 0.55) * Math.PI * 26)) * 12 : 0;

      const cb = getRotatedBoneBounds(cannonPivotX + cannonRecoil, cannonPivotY, CONFIG.mech.cannonLen, 16, aimAngleDeg);
      cannonX.push({ normalizedTime: t, value: { number: cb.left } });
      cannonY.push({ normalizedTime: t, value: { number: cb.top } });
      cannonRot.push({ normalizedTime: t, value: { number: cb.rotation } });

      // Muzzle Flash
      const muzzleTipX = cannonPivotX + Math.cos(toRad(aimAngleDeg)) * CONFIG.mech.cannonLen;
      const muzzleTipY = cannonPivotY + Math.sin(toRad(aimAngleDeg)) * CONFIG.mech.cannonLen;
      flashX.push({ normalizedTime: t, value: { number: round2(muzzleTipX - 13) } });
      flashY.push({ normalizedTime: t, value: { number: round2(muzzleTipY - 9) } });
      flashScale.push({ normalizedTime: t, value: { number: isMechFiring ? 1.4 : 0.0 } });
      flashOpacity.push({ normalizedTime: t, value: { number: isMechFiring ? 1.0 : 0.0 } });

      // ----------------------------------------------------------------------
      // STEP 3: DRONE LASER STRAFE & MECH ENERGY SHIELD INTERACTION
      // ----------------------------------------------------------------------
      const isDroneFiring = (t >= 0.32 && t <= 0.48);
      const shieldCenterPos = { x: currentX + 38, y: hullCenterY - 6 };

      // Drone Laser Beam targets Shield
      const laserDx = shieldCenterPos.x - dronePose.x;
      const laserDy = shieldCenterPos.y - dronePose.y;
      const laserDist = Math.hypot(laserDx, laserDy);
      const laserAngle = toDeg(Math.atan2(laserDy, laserDx));

      laserX.push({ normalizedTime: t, value: { number: round2((dronePose.x + shieldCenterPos.x) / 2 - 70) } });
      laserY.push({ normalizedTime: t, value: { number: round2((dronePose.y + shieldCenterPos.y) / 2 - 1.5) } });
      laserRot.push({ normalizedTime: t, value: { number: round2(laserAngle) } });
      laserScaleX.push({ normalizedTime: t, value: { number: round2(laserDist / 140) } });
      laserOpacity.push({ normalizedTime: t, value: { number: isDroneFiring ? 1.0 : 0.0 } });

      // Shield Deploys during Drone Fire
      const isShieldActive = (t >= 0.30 && t <= 0.52);
      shieldX.push({ normalizedTime: t, value: { number: round2(shieldCenterPos.x - 34) } });
      shieldY.push({ normalizedTime: t, value: { number: round2(shieldCenterPos.y - 42) } });
      shieldScale.push({ normalizedTime: t, value: { number: isShieldActive ? 1.2 + Math.sin(t * Math.PI * 30) * 0.1 : 0.0 } });
      shieldOpacity.push({ normalizedTime: t, value: { number: isShieldActive ? 0.9 : 0.0 } });

      // Shield Deflection Sparks
      sparksX.push({ normalizedTime: t, value: { number: round2(shieldCenterPos.x - 11) } });
      sparksY.push({ normalizedTime: t, value: { number: round2(shieldCenterPos.y - 11) } });
      sparksScale.push({ normalizedTime: t, value: { number: isDroneFiring ? 1.3 + Math.sin(t * Math.PI * 40) * 0.3 : 0.0 } });
      sparksOpacity.push({ normalizedTime: t, value: { number: isDroneFiring ? 1.0 : 0.0 } });

      // ----------------------------------------------------------------------
      // STEP 4: GUIDED MISSILE LAUNCH & DRONE FLARES
      // ----------------------------------------------------------------------
      let mX = currentX - 30, mY = hullCenterY - 26, mRot = -45, mOp = 0;
      if (t >= 0.75) {
        const u = (t - 0.75) / 0.25;
        mOp = 1.0;
        mX = (currentX - 30) + u * (dronePose.x - (currentX - 30));
        mY = (hullCenterY - 26) + u * (dronePose.y - (hullCenterY - 26)) - Math.sin(u * Math.PI) * 40;
        mRot = toDeg(Math.atan2(dronePose.y - mY, dronePose.x - mX));
      }
      missileX.push({ normalizedTime: t, value: { number: round2(mX - 9) } });
      missileY.push({ normalizedTime: t, value: { number: round2(mY - 3) } });
      missileRot.push({ normalizedTime: t, value: { number: round2(mRot) } });
      missileOpacity.push({ normalizedTime: t, value: { number: mOp } });

      // Drone Countermeasure Flare (t = 0.80 -> 0.95)
      const isFlare = (t >= 0.80 && t <= 0.95);
      flareX.push({ normalizedTime: t, value: { number: round2(dronePose.x + 20) } });
      flareY.push({ normalizedTime: t, value: { number: round2(dronePose.y + 10) } });
      flareScale.push({ normalizedTime: t, value: { number: isFlare ? 1.4 : 0.0 } });
      flareOpacity.push({ normalizedTime: t, value: { number: isFlare ? 0.9 : 0.0 } });

      // ----------------------------------------------------------------------
      // STEP 5: MECH DIGITIGRADE LEGS & SHADOW
      // ----------------------------------------------------------------------
      const footFar = getMechFootTrajectory(walkPhase + Math.PI, currentX);
      const ikFar = solveReverseLegIK({ x: currentX, y: currentHipY }, footFar, CONFIG.mech.thighLen, CONFIG.mech.shinLen);
      const thFb = getRotatedBoneBounds(currentX, currentHipY, CONFIG.mech.thighLen, 18, ikFar.thighAngleDeg);
      legL.thX.push({ normalizedTime: t, value: { number: thFb.left } });
      legL.thY.push({ normalizedTime: t, value: { number: thFb.top } });
      legL.thRot.push({ normalizedTime: t, value: { number: thFb.rotation } });
      legL.kneeX.push({ normalizedTime: t, value: { number: round2(ikFar.kneePos.x - 10) } });
      legL.kneeY.push({ normalizedTime: t, value: { number: round2(ikFar.kneePos.y - 10) } });
      const shFb = getRotatedBoneBounds(ikFar.kneePos.x, ikFar.kneePos.y, CONFIG.mech.shinLen, 14, ikFar.shinAngleDeg);
      legL.shX.push({ normalizedTime: t, value: { number: shFb.left } });
      legL.shY.push({ normalizedTime: t, value: { number: shFb.top } });
      legL.shRot.push({ normalizedTime: t, value: { number: shFb.rotation } });
      const ftFb = getRotatedBoneBounds(footFar.x - 12, footFar.y, 40, 12, footFar.rot);
      legL.ftX.push({ normalizedTime: t, value: { number: ftFb.left } });
      legL.ftY.push({ normalizedTime: t, value: { number: ftFb.top } });
      legL.ftRot.push({ normalizedTime: t, value: { number: ftFb.rotation } });

      const footNear = getMechFootTrajectory(walkPhase, currentX);
      const ikNear = solveReverseLegIK({ x: currentX, y: currentHipY }, footNear, CONFIG.mech.thighLen, CONFIG.mech.shinLen);

      legR.thX.push({ normalizedTime: t, value: { number: thNb.left } });
      legR.thY.push({ normalizedTime: t, value: { number: thNb.top } });
      legR.thRot.push({ normalizedTime: t, value: { number: thNb.rotation } });
      legR.kneeX.push({ normalizedTime: t, value: { number: round2(ikNear.kneePos.x - 11) } });
      legR.kneeY.push({ normalizedTime: t, value: { number: round2(ikNear.kneePos.y - 11) } });
      const shNb = getRotatedBoneBounds(ikNear.kneePos.x, ikNear.kneePos.y, CONFIG.mech.shinLen, 14, ikNear.shinAngleDeg);
      legR.shX.push({ normalizedTime: t, value: { number: shNb.left } });
      legR.shY.push({ normalizedTime: t, value: { number: shNb.top } });
      legR.shRot.push({ normalizedTime: t, value: { number: shNb.rotation } });
      const ftNb = getRotatedBoneBounds(footNear.x - 12, footNear.y, 42, 14, footNear.rot);
      legR.ftX.push({ normalizedTime: t, value: { number: ftNb.left } });
      legR.ftY.push({ normalizedTime: t, value: { number: ftNb.top } });
      legR.ftRot.push({ normalizedTime: t, value: { number: ftNb.rotation } });

      shadowX.push({ normalizedTime: t, value: { number: round2(currentX - 60) } });
      shadowY.push({ normalizedTime: t, value: { number: CONFIG.mech.groundY + 3 } });
      shadowScaleX.push({ normalizedTime: t, value: { number: round2(1.0 - bounceY / 16) } });
    }

    // ========================================================================
    // BIND CHARACTER LAYERS TO RUNTIME NODES
    // ========================================================================
    // 1. Mech Far Leg
    layers.push({ runtimeNodeId: ids.get("mech_leg_L_thigh"), properties: [makeTrack("PositionX", legL.thX), makeTrack("PositionY", legL.thY), makeTrack("Rotation", legL.thRot)] });

    layers.push({ runtimeNodeId: ids.get("mech_leg_L_foot"),  properties: [makeTrack("PositionX", legL.ftX), makeTrack("PositionY", legL.ftY), makeTrack("Rotation", legL.ftRot)] });

    // 2. Mech Torso, Weapons, Shield
    layers.push({ runtimeNodeId: ids.get("mech_shadow"),         properties: [makeTrack("PositionX", shadowX), makeTrack("PositionY", shadowY), makeTrack("ScaleX", shadowScaleX)] });
